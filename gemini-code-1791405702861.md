# Feature Specification: Video Scrollytelling Section (Next.js)

## 1. Overview
Build a high-performance, responsive scrollytelling section in a Next.js (App Router) application. The component synchronizes vertical page scrolling with a sticky video of a desk workstation pan, revealing narrative cards aligned with physical objects in the frame[cite: 1].

---

## 2. Tech Stack & Dependencies
- **Framework**: Next.js 14+ (App Router, TypeScript)
- **Styling**: Tailwind CSS
- **Animation**: `framer-motion`
- **Asset**: `public/desk-pan.mp4` (pan across coffee, laptop DAW, desk lamp/notes, and architectural blueprint[cite: 1])

## 4. Technical Requirements

### 4.1. Layout & Scroll Track
- **Container**: Track height must be set to `h-[450vh]` to give adequate scroll distance for scrub control.
- **Sticky Viewport**: A `sticky top-0 h-screen w-full` wrapper holding the `<video>` element with `object-cover`.
- **Overlay**: Subtle radial and linear dark gradients over the video to ensure text legibility while preserving visibility of the subject.

### 4.2. Video Scrub Synchronization
- The `<video>` element must have `playsInline`, `muted`, and `preload="auto"` enabled.
- Track scroll progress using Framer Motion's `useScroll` (`offset: ["start start", "end end"]`).
- Update `videoRef.current.currentTime` dynamically inside an optimized `requestAnimationFrame` loop to prevent DOM thrashing and jank.
- Ensure duration is read only after `onLoadedMetadata` triggers to prevent `NaN` assignments.

### 4.3. Narrative Card Interactions
- Each card must trigger sequentially based on its scroll interval (fade-in, hold, fade-out).
- Use `useTransform` to bind card `opacity` and vertical translation `y` to the overall container scroll progress.
- Floating cards must feature glassmorphic styling: semi-transparent dark background (`bg-black/60` or `bg-slate-950/65`), `backdrop-blur-xl`, subtle border `border-white/10`, and rounded corners (`rounded-2xl`).
- Container wrappers must have `pointer-events-none`, while the interactive cards retain `pointer-events-auto`.

## 5. File Structure & Implementation

### File: `components/DeskScrollytelling.tsx`
```tsx
"use client";

import { useEffect, useRef, useState } from "react";
import { motion, useScroll, useTransform, MotionValue } from "framer-motion";

interface Step {
  id: number;
  tag: string;
  title: string;
  description: string;
  align: "left" | "right";
}

const steps: Step[] = [
  {
    id: 1,
    tag: "Phase 01",
    title: "The Initial Spark",
    description:
      "Every project starts in silence—warm coffee, an open notepad, and ideas waiting to take form.",
    align: "left",
  },
  {
    id: 2,
    tag: "Phase 02",
    title: "Audio Engineering",
    description:
      "Transforming intangible concepts into dynamic waveforms, frequencies, and rhythm.",
    align: "right",
  },
  {
    id: 3,
    tag: "Phase 03",
    title: "Deep Focus & Iteration",
    description:
      "Refining the details under late-night lamplight, balancing theory with handwritten precision.",
    align: "left",
  },
  {
    id: 4,
    tag: "Phase 04",
    title: "The Blueprint",
    description:
      "Grounding acoustics and ideas into structural layouts and tangible reality.",
    align: "right",
  },
];

export default function DeskScrollytelling() {
  const containerRef = useRef<HTMLDivElement>(null);
  const videoRef = useRef<HTMLVideoElement>(null);
  const [duration, setDuration] = useState(0);

  const { scrollYProgress } = useScroll({
    target: containerRef,
    offset: ["start start", "end end"],
  });

  const handleLoadedMetadata = () => {
    if (videoRef.current) {
      setDuration(videoRef.current.duration);
    }
  };

  useEffect(() => {
    let frameId: number;

    const unsubscribe = scrollYProgress.on("change", (latest) => {
      if (!videoRef.current || duration === 0) return;

      frameId = requestAnimationFrame(() => {
        if (videoRef.current) {
          videoRef.current.currentTime = Math.min(
            Math.max(latest * duration, 0),
            duration
          );
        }
      });
    });

    return () => {
      unsubscribe();
      cancelAnimationFrame(frameId);
    };
  }, [scrollYProgress, duration]);

  return (
    <section ref={containerRef} className="relative h-[450vh] bg-black">
      {/* Pinned Video Layer */}
      <div className="sticky top-0 h-screen w-full overflow-hidden flex items-center justify-center">
        <video
          ref={videoRef}
          src="/desk-pan.mp4"
          playsInline
          muted
          preload="auto"
          onLoadedMetadata={handleLoadedMetadata}
          className="absolute inset-0 h-full w-full object-cover filter brightness-[0.8] contrast-[1.05]"
        />
        <div className="absolute inset-0 bg-gradient-to-t from-black/80 via-transparent to-black/40 pointer-events-none" />
        <div className="absolute inset-0 bg-radial-[circle_at_center,transparent_40%,rgba(0,0,0,0.8)_100%] pointer-events-none" />
      </div>

      {/* Narrative Overlay Track */}
      <div className="relative -mt-[100vh] z-10 flex flex-col pointer-events-none">
        {steps.map((step, idx) => (
          <StoryCard index="{idx}" key="{step.id}" scrollYProgress="{scrollYProgress}" step="{step}"/>
        ))}
      </div>
    </section>
  );
}

function StoryCard({
  step,
  index,
  scrollYProgress,
}: {
  step: Step;
  index: number;
  scrollYProgress: MotionValue<number>;
}) {
  const start = index * 0.25;
  const peak = start + 0.12;
  const end = (index + 1) * 0.25;

  const opacity = useTransform(scrollYProgress, [start, peak, end], [0, 1, 0]);
  const translateY = useTransform(scrollYProgress, [start, peak, end], [40, 0, -40]);

  return (
    <div className="h-screen w-full flex items-center px-6 md:px-24">
      <div
        className={`w-full flex ${
          step.align === "right" ? "justify-end" : "justify-start"
        }`}
      >
        <motion.div
          style={{ opacity, y: translateY }}
          className="max-w-md w-full rounded-2xl border border-white/10 bg-black/60 p-6 md:p-8 backdrop-blur-xl shadow-2xl pointer-events-auto"
        >
          <span className="inline-block text-xs font-semibold tracking-widest uppercase text-amber-400 mb-2">
            {step.tag}
          </span>
          <h3 className="text-2xl font-bold tracking-tight text-white mb-2">
            {step.title}
          </h3>
          <p className="text-sm md:text-base leading-relaxed text-zinc-300">
            {step.description}
          </p>
        </motion.div>
      </div>
    </div>
  );
}
